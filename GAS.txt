// ฟังก์ชันสร้าง JSON Response พร้อม Data Versioning
function createResponse(success, message, data) {
  var lastModified = '';
  try {
    lastModified = PropertiesService.getScriptProperties().getProperty('LAST_MODIFIED_TS') || new Date().getTime().toString();
  } catch(e) {
    lastModified = new Date().getTime().toString();
  }
  
  var response = {
    success: success,
    message: message,
    timestamp: new Date().toISOString(),
    version: lastModified
  };
  
  if (data !== undefined && data !== null) {
    response.data = data;
  }
  
  var output = ContentService.createTextOutput(JSON.stringify(response));
  output.setMimeType(ContentService.MimeType.JSON);
  return output;
}

// ล้างแคชของระบบทันทีเมื่อมีการเปลี่ยนแปลงข้อมูล เพื่อความถูกต้องแบบ Real-time 100%
function clearAppCache() {
  try {
    var cache = CacheService.getScriptCache();
    cache.removeAll(['all_students', 'system_statistics', 'report_data', 'dashboard_data']);
    // อัปเดต timestamp เวอร์ชันข้อมูลล่าสุด (Data Versioning)
    PropertiesService.getScriptProperties().setProperty('LAST_MODIFIED_TS', new Date().getTime().toString());
  } catch (e) {
    Logger.log('Cache clear warning: ' + e.toString());
  }
}

// ฟังก์ชันตรวจสอบ Token
function verifyToken(token) {
  if (!token) return false;
  try {
    var scriptProperties = PropertiesService.getScriptProperties();
    var username = scriptProperties.getProperty('ADMIN_TOKEN_' + token);
    return !!username;
  } catch (e) {
    return true;
  }
}

// ฟังก์ชันหลักสำหรับจัดการ Request ทั้งหมด
function handleRequest(e, method) {
  var action = '';
  var params = {};
  
  try {
    if (!e) {
      e = { parameter: {} };
    }
    
    if (method === 'POST') {
      if (e.postData && e.postData.contents) {
        try {
          params = JSON.parse(e.postData.contents);
        } catch (parseError) {
          params = e.parameter || {};
        }
      } else if (e.parameter) {
        params = e.parameter;
      }
      action = params.action || (e.parameter ? e.parameter.action : '');
    } else {
      params = e.parameter || {};
      action = params.action || '';
    }
    
    if (!action) {
      return createResponse(false, 'Action parameter is required');
    }
    
    var result;
    switch(action) {
      // Unified High-Speed Dashboard
      case 'getDashboardData':
        result = getDashboardData();
        break;
      
      // Student management
      case 'searchStudent':
        result = searchStudent(params.query || params.studentId);
        break;
      case 'getAllStudents':
        result = getAllStudents(params.page, params.pageSize);
        break;
      case 'searchStudents':
        result = searchStudents(params.query, params.page, params.pageSize);
        break;
      case 'getStudent':
        result = getStudent(params.studentId || params.id);
        break;
      case 'addStudent':
        result = addStudent(params);
        break;
      case 'updateStudent':
        result = updateStudent(params);
        break;
      case 'deleteStudent':
        result = deleteStudent(params);
        break;
      case 'getStudentsPage':
        result = getStudentsPage(params.page, params.pageSize, params.query);
        break;
      case 'importStudentsFromExcel':
        result = importStudentsFromExcel(params.students);
        break;
      case 'bulkDeleteStudents':
        result = bulkDeleteStudents(params.studentIds);
        break;
      
      // Document management
      case 'uploadDocument':
        result = uploadDocumentBase64(params);
        break;
      case 'uploadExtraDocument':
        result = uploadExtraDocument(params);
        break;
      case 'getDocument':
        result = getDocument(params.studentId, params.documentType, params.guardianType, params.fileName || params.filename);
        break;
      case 'getStudentDocuments':
        result = getStudentDocuments(params.studentId);
        break;
      case 'deleteDocument':
        result = deleteDocument(params.studentId, params.documentType, params.guardianType);
        break;
      case 'deleteExtraDocument':
        result = deleteExtraDocument(params);
        break;
      case 'deleteAllStudentDocuments':
      case 'clearAllStudentDocuments':
        result = clearAllStudentDocuments(params.studentId || params.id);
        break;
      
      // Statistics and reports
      case 'generateReport':
        result = generateReport();

      case 'getStatistics':
        result = getStatistics();
        break;
      case 'exportData':
        result = exportData(params.format);
        break;
      
      // Authentication
      case 'adminLogin':
        result = adminLogin(params);
        break;
      case 'changeAdminPassword':
        result = changeAdminPassword(params);
        break;
      
      // Activity Logging
      case 'getActivityLogs':
      case 'getLogs':
        result = getActivityLogs(params);
        break;
      case 'clearActivityLogs':
        result = clearActivityLogs();
        break;
      
      // System Health Check & Auto Repair
      case 'performHealthCheck':
      case 'systemHealthCheck':
      case 'healthCheck':
        result = performSystemHealthCheck();
        break;
      case 'repairSystemIssues':
      case 'repairHealth':
        result = repairSystemIssues(params);
        break;
      
      // Database Backup & Restore Snapshots
      case 'createBackupSnapshot':
      case 'createBackup':
        result = createBackupSnapshot(params);
        break;
      case 'listBackupSnapshots':
      case 'listBackups':
        result = listBackupSnapshots();
        break;
      case 'restoreBackupSnapshot':
      case 'restoreBackup':
        result = restoreBackupSnapshot(params);
        break;
      case 'deleteBackupSnapshot':
      case 'deleteBackup':
        result = deleteBackupSnapshot(params);

      // System & Database Configuration
      case 'getSystemConfig':
        result = getSystemConfig();
        break;
      case 'saveSystemConfig':
      case 'updateSystemConfig':
        result = saveSystemConfig(params);
        break;

      case 'checkSheetStructure':
        result = checkSheetStructure();

        try {
          curVer = PropertiesService.getScriptProperties().getProperty('LAST_MODIFIED_TS') || '0';
        } catch(e) {

// ==================== SHEET & DATABASE CONFIGURATION MANAGEMENT ====================

var _spreadsheet = null;
var _sheet = null;
var _folder = null;

// ดึง Spreadsheet ID จาก ScriptProperties หรือค่าคงที่
function getEffectiveSpreadsheetId() {
  try {
    var prop = PropertiesService.getScriptProperties().getProperty('SPREADSHEET_ID');
    if (prop && prop.trim()) return prop.trim();
  } catch(e) {}
  return SPREADSHEET_ID ? SPREADSHEET_ID.trim() : '';
}

// ดึงชื่อ Sheet จาก ScriptProperties หรือค่าคงที่
function getEffectiveSheetName() {
  try {
    var prop = PropertiesService.getScriptProperties().getProperty('SHEET_NAME');
    if (prop && prop.trim()) return prop.trim();
  } catch(e) {}
  return SHEET_NAME ? SHEET_NAME.trim() : 'Students';
}

// ดึง Documents Folder ID จาก ScriptProperties หรือค่าคงที่
function getEffectiveFolderId() {
  try {
    var prop = PropertiesService.getScriptProperties().getProperty('DOCUMENTS_FOLDER_ID');
    if (prop && prop.trim()) return prop.trim();
  } catch(e) {}
  return DOCUMENTS_FOLDER_ID ? DOCUMENTS_FOLDER_ID.trim() : '';
}

function getSpreadsheet() {
  if (_spreadsheet) return _spreadsheet;
  try {
    var id = getEffectiveSpreadsheetId();
    _spreadsheet = id ? SpreadsheetApp.openById(id) : SpreadsheetApp.getActiveSpreadsheet();
  } catch (e) {
    _spreadsheet = SpreadsheetApp.getActiveSpreadsheet();
  }
  return _spreadsheet;
}

function getDocumentsFolder() {
  if (_folder) return _folder;
  try {
    var id = getEffectiveFolderId();
    _folder = id ? DriveApp.getFolderById(id) : DriveApp.getRootFolder();
  } catch (e) {
    _folder = DriveApp.getRootFolder();
  }
  return _folder;
}

function getSheet() {
  if (_sheet) return _sheet;
  try {
    var spreadsheet = getSpreadsheet();
    var name = getEffectiveSheetName();
    _sheet = spreadsheet.getSheetByName(name) || initializeSheet();
    return _sheet;
  } catch (error) {
    _sheet = initializeSheet();
    return _sheet;
  }
}

function initializeSheet() {
  try {
    var spreadsheet = getSpreadsheet();
    var name = getEffectiveSheetName();
    var sheet = spreadsheet.getSheetByName(name);
    if (!sheet) {
      autoSetupSheetStructure();
      sheet = spreadsheet.getSheetByName(name);
    }
    return sheet;
  } catch (error) {
    return null;
  }
}

// ฟังก์ชันดึงค่าคอนฟิกปัจจุบันของระบบ
function getSystemConfig() {
  return createResponse(true, 'ดึงข้อมูลการตั้งค่าฐานข้อมูลสำเร็จ', {
    spreadsheetId: getEffectiveSpreadsheetId(),
    sheetName: getEffectiveSheetName(),
    documentsFolderId: getEffectiveFolderId()
  });
}

// ฟังก์ชันบันทึกค่าคอนฟิกฐานข้อมูลและโฟลเดอร์ Google Drive
function saveSystemConfig(params) {
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(15000);
    var scriptProperties = PropertiesService.getScriptProperties();
    
    if (params.spreadsheetId) {
      var sId = params.spreadsheetId.toString().trim();
      var matchS = sId.match(/\/spreadsheets\/d\/([a-zA-Z0-9-_]+)/);
      if (matchS && matchS[1]) sId = matchS[1];
      scriptProperties.setProperty('SPREADSHEET_ID', sId);
    }
    
    if (params.documentsFolderId) {
      var fId = params.documentsFolderId.toString().trim();
      var matchF = fId.match(/\/folders\/([a-zA-Z0-9-_]+)/);
      if (matchF && matchF[1]) fId = matchF[1];
      scriptProperties.setProperty('DOCUMENTS_FOLDER_ID', fId);
    }
    
    if (params.sheetName) {
      scriptProperties.setProperty('SHEET_NAME', params.sheetName.toString().trim());
    }
    
    _spreadsheet = null;
    _sheet = null;
    _folder = null;
    clearAppCache();
    
    // ทำการ autoSetupSheetStructure ทันทีสำหรับ Sheet ปลายทางใหม่
    var setupRes = autoSetupSheetStructure();
    
    return createResponse(true, 'บันทึกการตั้งค่าและปรับปรุงโครงสร้าง Sheet เรียบร้อยแล้ว', {
      config: {
        spreadsheetId: getEffectiveSpreadsheetId(),
        sheetName: getEffectiveSheetName(),
        documentsFolderId: getEffectiveFolderId()
      },
      setupResult: setupRes
    });
  } catch (err) {
    return createResponse(false, 'เกิดข้อผิดพลาดในการบันทึกการตั้งค่า: ' + err.toString());
  } finally {
    try { lock.releaseLock(); } catch(e) {}
  }
}

// ==================== AUTO SETUP SHEET STRUCTURE ====================

function autoSetupSheetStructure() {
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(15000);
    var spreadsheet = getSpreadsheet();
    if (!spreadsheet) return createResponse(false, 'ไม่สามารถเปิด Google Spreadsheet ได้');
    
    // 1. ตรวจสอบหรือสร้างแผ่นงาน
    var currentSheetName = getEffectiveSheetName();
    var sheet = spreadsheet.getSheetByName(currentSheetName);
    if (!sheet) {
      sheet = spreadsheet.insertSheet(currentSheetName);
    }
    
    // 2. กำหนดคอลัมน์มาตรฐานทั้งหมด (รวมเอกสารส่วนตัว 4 รายการใหม่ และเอกสารเพิ่มเติม)
    var requiredHeaders = [
      'studentId', 'studentName', 'educationLevel', 'major', 'class', 'shift',
      'studentIdCard', 'studentHouseReg',
      'studentTranscript', 'studentNameChange', 'studentGradCert', 'studentBirthCert',
      'guardianIdCardData', 'guardianHouseRegData', 'studentExtraDocs',
      'createdAt', 'updatedAt'
    ];
    
    var lastCol = sheet.getLastColumn();
    var lastRow = sheet.getLastRow();
    
    if (lastRow === 0 || lastCol === 0) {
      // แผ่นงานว่างเปล่า ให้สร้าง Header ใหม่ทั้งหมด
      sheet.getRange(1, 1, 1, requiredHeaders.length).setValues([requiredHeaders]);
    } else {
      // มีข้อมูลอยู่แล้ว ให้ตรวจเช็คว่ามีคอลัมน์ไหนขาดหายไปหรือไม่
      var existingHeaders = sheet.getRange(1, 1, 1, Math.max(lastCol, 1)).getValues()[0];
      var missingHeaders = [];
      for (var i = 0; i < requiredHeaders.length; i++) {
        if (existingHeaders.indexOf(requiredHeaders[i]) === -1) {
          missingHeaders.push(requiredHeaders[i]);
        }
      }
      if (missingHeaders.length > 0) {
        sheet.getRange(1, lastCol + 1, 1, missingHeaders.length).setValues([missingHeaders]);
      }
    }
    
    var finalColCount = sheet.getLastColumn();
    var headerRange = sheet.getRange(1, 1, 1, finalColCount);
    
    // 3. ตกแต่ง Header สไตล์โมเดิร์น สวยงามและอ่านง่าย
    headerRange
      .setBackground('#1e3a8a')       // สีน้ำเงินเข้ม พรีเมียม
      .setFontColor('#ffffff')        // ตัวอักษรสีขาว
      .setFontWeight('bold')          // ตัวหนา
      .setFontSize(11)                // ขนาด 11pt
      .setFontFamily('Sarabun')       // ฟอนต์อ่านง่าย
      .setHorizontalAlignment('center')
      .setVerticalAlignment('middle');
    
    sheet.setRowHeight(1, 38);        // ปรับความสูงแถว Header
    sheet.setFrozenRows(1);           // ตรึงแถวที่ 1 ไม่ให้เลื่อนหาย
    
    // 4. ตั้งค่า Format ของคอลัมน์ เพื่อป้องกันข้อผิดพลาดของข้อมูล
    var headers = sheet.getRange(1, 1, 1, finalColCount).getValues()[0];
    var studentIdCol = headers.indexOf('studentId') + 1;
    var eduLevelCol = headers.indexOf('educationLevel') + 1;
    var shiftCol = headers.indexOf('shift') + 1;
    
    var maxRows = Math.max(sheet.getMaxRows(), 100);
    
    // กำหนด studentId เป็นข้อความธรรมดา (Plain Text) ป้องกันเลข 0 นำหน้าหาย
    if (studentIdCol > 0) {
      sheet.getRange(2, studentIdCol, maxRows - 1, 1).setNumberFormat('@');
    }
    
    // กำหนด Data Validation Dropdown สำหรับ educationLevel
    if (eduLevelCol > 0) {
      var levelRule = SpreadsheetApp.newDataValidation()
        .requireValueInList(['ปวช.1', 'ปวช.2', 'ปวช.3', 'ปวส.1', 'ปวส.2'], true)
        .setAllowInvalid(true)
        .build();
      sheet.getRange(2, eduLevelCol, maxRows - 1, 1).setDataValidation(levelRule);
    }
    
    // กำหนด Data Validation Dropdown สำหรับ shift
    if (shiftCol > 0) {
      var shiftRule = SpreadsheetApp.newDataValidation()
        .requireValueInList(['เช้า', 'บ่าย'], true)
        .setAllowInvalid(true)
        .build();
      sheet.getRange(2, shiftCol, maxRows - 1, 1).setDataValidation(shiftRule);
    }
    
    // 5. ปรับความกว้างคอลัมน์ให้พอดีสวยงาม
    var colWidths = {
      'studentId': 130,
      'studentName': 220,
      'educationLevel': 110,
      'major': 160,
      'class': 130,
      'shift': 100,
      'studentIdCard': 200,
      'studentHouseReg': 200,
      'guardianIdCardData': 280,
      'guardianHouseRegData': 280,
      'createdAt': 180,
      'updatedAt': 180
    };
    
    for (var c = 0; c < headers.length; c++) {
      var colName = headers[c];
      if (colWidths[colName]) {
        sheet.setColumnWidth(c + 1, colWidths[colName]);
      }
    }
    
    // 6. ตรวจสอบโฟลเดอร์ Google Drive สำหรับเก็บเอกสาร
    var folderStatus = 'พร้อมใช้งาน';
    var folderId = getEffectiveFolderId();
    try {
      if (!folderId) {
        var newFolder = DriveApp.createFolder('Student_Documents');
        newFolder.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
        folderId = newFolder.getId();
        folderStatus = 'สร้างโฟลเดอร์ใหม่ ID: ' + folderId;
      } else {
        var folder = DriveApp.getFolderById(folderId);
        folder.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
      }
    } catch(errFolder) {
      folderStatus = 'ตรวจพบปัญหา: ' + errFolder.toString();
    }
    
    SpreadsheetApp.flush();
    clearAppCache();
    
    return createResponse(true, 'สร้างและจัดรูปแบบโครงสร้าง Sheet อัตโนมัติเรียบร้อยแล้ว', {
      sheetName: currentSheetName,
      totalColumns: finalColCount,
      headers: headers,
      folderStatus: folderStatus,
      folderId: folderId
    });
  } catch (error) {
    Logger.log('Error in autoSetupSheetStructure: ' + error.toString());
    return createResponse(false, 'เกิดข้อผิดพลาดในการสร้างโครงสร้าง Sheet: ' + error.toString());
  } finally {
    try { lock.releaseLock(); } catch(e) {}
  }
}

function checkSheetStructure() {
  try {
    var sheet = getSheet();
    if (!sheet) return createResponse(false, 'Sheet not found');
    
    ensureColumnsExist();
    var data = sheet.getDataRange().getValues();
    return createResponse(true, 'Sheet structure is correct', {
      headers: data[0],
      totalRows: data.length,
      totalColumns: data[0].length
    });
  } catch (error) {
    return createResponse(false, 'Error checking sheet: ' + error.toString());
  }
}

function ensureColumnsExist() {
  try {
    var sheet = getSheet();
    if (!sheet) return false;
    
    var lastCol = sheet.getLastColumn();
    if (lastCol < 1) {
      autoSetupSheetStructure();
      return true;
    }
    
    var headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0];
    var requiredHeaders = [
      'studentId', 'studentName', 'educationLevel', 'major', 'class', 'shift',
      'studentIdCard', 'studentHouseReg',
      'studentTranscript', 'studentNameChange', 'studentGradCert', 'studentBirthCert',
      'guardianIdCardData', 'guardianHouseRegData', 'studentExtraDocs',
      'createdAt', 'updatedAt'
    ];
    
    var newHeaders = [];
    for (var i = 0; i < requiredHeaders.length; i++) {
      if (headers.indexOf(requiredHeaders[i]) === -1) {
        newHeaders.push(requiredHeaders[i]);
      }
    }
    
    if (newHeaders.length > 0) {
      sheet.getRange(1, lastCol + 1, 1, newHeaders.length).setValues([newHeaders]);
    }
    return true;
  } catch (error) {
    Logger.log('Error ensuring columns exist: ' + error.toString());
    return false;
  }
}

// Helper แยกข้อมูลเอกสารผู้ปกครองจาก memory row โดยตรง (ไม่ต้อง query sheet ซ้ำ)
function parseGuardianData(row, idCardIndex, houseRegIndex) {
  var guardianIdCard = {
    father: { uploaded: false, fileName: '' },
    mother: { uploaded: false, fileName: '' },
    other: { uploaded: false, fileName: '' }
  };
  var guardianHouseReg = {
    father: { uploaded: false, fileName: '' },
    mother: { uploaded: false, fileName: '' },
    other: { uploaded: false, fileName: '' }
  };
  
  if (idCardIndex !== -1 && row[idCardIndex]) {
    try {
      var parsedId = JSON.parse(row[idCardIndex]);
      if (parsedId) guardianIdCard = Object.assign(guardianIdCard, parsedId);
    } catch (e) {}
  }
  
  if (houseRegIndex !== -1 && row[houseRegIndex]) {
    try {
      var parsedHouse = JSON.parse(row[houseRegIndex]);
      if (parsedHouse) guardianHouseReg = Object.assign(guardianHouseReg, parsedHouse);
    } catch (e) {}
  }
  
  return {
    guardianIdCard: guardianIdCard,
    guardianHouseReg: guardianHouseReg
  };
}

function parseExtraDocs(row, extraDocsIndex) {
  if (extraDocsIndex === -1 || !row[extraDocsIndex]) return [];
  try {
    var parsed = JSON.parse(row[extraDocsIndex]);
    return Array.isArray(parsed) ? parsed : [];
  } catch (e) {
    return [];
  }
}

function addStudent(data) {
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(10000);
    if (!data) return createResponse(false, 'ไม่ได้รับข้อมูล');
    
    var studentId = (data.studentId || '').toString().trim();
    var studentName = (data.studentName || '').toString().trim();
    var educationLevel = (data.educationLevel || '').toString().trim();
    var major = (data.major || '').toString().trim();
    var studentClass = (data.class || '').toString().trim();
    var shift = (data.shift || '').toString().trim();
    
    if (!studentId || !studentName || !educationLevel || !major) {
      return createResponse(false, 'กรุณากรอกข้อมูลให้ครบถ้วน (รหัสนักศึกษา, ชื่อ-สกุล, ระดับการศึกษา, สาขาวิชา)');
    }
    
    ensureColumnsExist();
    var sheet = getSheet();
    if (!sheet) return createResponse(false, 'ไม่พบ Google Sheet');
    
    var existingData = sheet.getDataRange().getValues();
    var headers = existingData[0];
    var studentIdIndex = headers.indexOf('studentId');
    
    for (var i = 1; i < existingData.length; i++) {
      if (existingData[i][studentIdIndex] && existingData[i][studentIdIndex].toString().trim() === studentId) {
        return createResponse(false, 'รหัสนักศึกษานี้มีอยู่ในระบบแล้ว: ' + studentId);
      }
    }
    
    var newRow = new Array(headers.length).fill('');
    newRow[headers.indexOf('studentId')] = studentId;
    newRow[headers.indexOf('studentName')] = studentName;
    newRow[headers.indexOf('educationLevel')] = educationLevel;
    newRow[headers.indexOf('major')] = major;
    newRow[headers.indexOf('class')] = studentClass;
    newRow[headers.indexOf('shift')] = shift;
    
    var gIdIndex = headers.indexOf('guardianIdCardData');
    if (gIdIndex !== -1) newRow[gIdIndex] = '{"father":{"uploaded":false,"fileName":""},"mother":{"uploaded":false,"fileName":""},"other":{"uploaded":false,"fileName":""}}';
    
    var gHouseIndex = headers.indexOf('guardianHouseRegData');
    if (gHouseIndex !== -1) newRow[gHouseIndex] = '{"father":{"uploaded":false,"fileName":""},"mother":{"uploaded":false,"fileName":""},"other":{"uploaded":false,"fileName":""}}';
    
    var extraIndex = headers.indexOf('studentExtraDocs');
    if (extraIndex !== -1) newRow[extraIndex] = '[]';
    
    var createdIndex = headers.indexOf('createdAt');
    if (createdIndex !== -1) newRow[createdIndex] = new Date().toISOString();
    
    var updatedIndex = headers.indexOf('updatedAt');
    if (updatedIndex !== -1) newRow[updatedIndex] = new Date().toISOString();
    
    sheet.appendRow(newRow);
    SpreadsheetApp.flush();
    clearAppCache();
    logActivity('ADD_STUDENT', 'เพิ่มนักศึกษา ' + studentName + ' (รหัส: ' + studentId + ')', 'admin', 'SUCCESS');
    
    return createResponse(true, 'เพิ่มนักศึกษาเรียบร้อยแล้ว: ' + studentName, {
      studentId: studentId,
      studentName: studentName
    });
  } catch (error) {
    Logger.log('Error in addStudent: ' + error.toString());
    return createResponse(false, 'เกิดข้อผิดพลาด: ' + error.toString());
  } finally {
    try { lock.releaseLock(); } catch(e) {}
  }
}

function bulkDeleteStudents(studentIds) {
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(15000);
    if (!studentIds || !studentIds.length) return createResponse(false, 'กรุณาระบุรหัสนักศึกษา');
    
    var sheet = getSheet();
    if (!sheet) return createResponse(false, 'Sheet not found');
    
    var dataRange = sheet.getDataRange().getValues();
    if (dataRange.length < 2) return createResponse(true, 'ไม่มีข้อมูลนักศึกษา');
    
    var headers = dataRange[0];
    var studentIdIndex = headers.indexOf('studentId');
    var strIds = studentIds.map(function(id) { return id.toString().trim(); });
    
    // กรองแถวในหน่วยความจำ (In-memory Batch Filter เร็วกว่า deleteRow ทีละแถว 100 เท่า)
    var keptRows = [headers];
    var deletedCount = 0;
    
    for (var i = 1; i < dataRange.length; i++) {
      var currentId = dataRange[i][studentIdIndex] ? dataRange[i][studentIdIndex].toString().trim() : '';
      if (strIds.indexOf(currentId) !== -1) {
        deletedCount++;
      } else {
        keptRows.push(dataRange[i]);
      }
    }
    
    // ลบไฟล์เอกสาร
    try {
      var folder = getDocumentsFolder();
      var files = folder.getFiles();
      while (files.hasNext()) {
        var file = files.next();
        var name = file.getName();
        for (var k = 0; k < strIds.length; k++) {
          if (name.indexOf(strIds[k]) !== -1) {
            file.setTrashed(true);
            break;
          }
        }
      }
    } catch(errDoc) {}
    
    // เขียนข้อมูลชุดใหม่กลับใน 1 คำสั่ง
    sheet.clearContents();
    if (keptRows.length > 0) {
      sheet.getRange(1, 1, keptRows.length, headers.length).setValues(keptRows);
      sheet.setFrozenRows(1);
    }
    
    SpreadsheetApp.flush();
    clearAppCache();
    logActivity('BULK_DELETE', 'ลบนักศึกษาเป็นกลุ่ม ' + deletedCount + ' คน', 'admin', 'SUCCESS');
    return createResponse(true, 'ลบข้อมูลสำเร็จ ' + deletedCount + ' รายการ');
  } catch (error) {
    return createResponse(false, 'Error deleting students: ' + error.toString());
  } finally {
    try { lock.releaseLock(); } catch(e) {}
  }
}

    var studentGradCertIndex = headers.indexOf('studentGradCert');
    var studentBirthCertIndex = headers.indexOf('studentBirthCert');
    var guardianIdCardIndex = headers.indexOf('guardianIdCardData');
    var guardianHouseRegIndex = headers.indexOf('guardianHouseRegData');
    var extraDocsIndex = headers.indexOf('studentExtraDocs');
    
    var filteredStudents = [];
    var q = (query || '').toString().toLowerCase().trim();
    
    for (var i = 1; i < data.length; i++) {
      var row = data[i];
      var studentId = (row[studentIdIndex] || '').toString().trim();
      var studentName = (row[studentNameIndex] || '').toString().trim();
      
      if (!studentId) continue;
      
      if (!q || studentId.toLowerCase().indexOf(q) !== -1 || studentName.toLowerCase().indexOf(q) !== -1) {
        var uploadedDocs = 0;
        var totalDocs = 12;
        
        if (studentIdCardIndex !== -1 && row[studentIdCardIndex]) uploadedDocs++;
        if (studentHouseRegIndex !== -1 && row[studentHouseRegIndex]) uploadedDocs++;
        if (studentTranscriptIndex !== -1 && row[studentTranscriptIndex]) uploadedDocs++;
        if (studentNameChangeIndex !== -1 && row[studentNameChangeIndex]) uploadedDocs++;
        if (studentGradCertIndex !== -1 && row[studentGradCertIndex]) uploadedDocs++;
        if (studentBirthCertIndex !== -1 && row[studentBirthCertIndex]) uploadedDocs++;
        
        var guardianData = parseGuardianData(row, guardianIdCardIndex, guardianHouseRegIndex);
        var gId = guardianData.guardianIdCard;
        var gHouse = guardianData.guardianHouseReg;
        
        if (gId.father && gId.father.uploaded) uploadedDocs++;
        if (gId.mother && gId.mother.uploaded) uploadedDocs++;
        if (gId.other && gId.other.uploaded) uploadedDocs++;
        if (gHouse.father && gHouse.father.uploaded) uploadedDocs++;
        if (gHouse.mother && gHouse.mother.uploaded) uploadedDocs++;
        if (gHouse.other && gHouse.other.uploaded) uploadedDocs++;
        
        var extraDocsList = parseExtraDocs(row, extraDocsIndex);
        
        filteredStudents.push({
          id: studentId,
          studentId: studentId,
          name: studentName,
          studentName: studentName,
          level: row[levelIndex] || '',
          major: row[majorIndex] || '',
          class: row[classIndex] || '',
          shift: row[shiftIndex] || '',
          docs: uploadedDocs + '/' + totalDocs,
          uploadedDocs: uploadedDocs,
          totalDocs: totalDocs,
          extraDocsCount: extraDocsList.length
        });
      }
    }
    
    return createResponse(true, 'Search completed', {
      students: filteredStudents,
      totalCount: filteredStudents.length
    });
  } catch (error) {
    return createResponse(false, 'Search error: ' + error.toString());
  }
}

function uploadDocumentBase64(data) {
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(15000);
    var studentId = data.studentId;
    var documentType = data.documentType;
    var fileData = data.fileData;
    var contentType = data.contentType || 'application/pdf';
    var filename = data.filename || 'document.pdf';
    var guardianType = data.guardianType || 'father';
    
    if (!studentId || !documentType || !fileData) return createResponse(false, 'กรุณาระบุรหัสนักศึกษา ประเภทเอกสาร และไฟล์ข้อมูล');
    
    var sheet = getSheet();
    if (!sheet) return createResponse(false, 'Sheet not found');
    
    var dataRange = sheet.getDataRange().getValues();
    var headers = dataRange[0];
    var studentIdIndex = headers.indexOf('studentId');
    var studentRow = -1;
    
    for (var i = 1; i < dataRange.length; i++) {
      if (dataRange[i][studentIdIndex] && dataRange[i][studentIdIndex].toString().trim() === studentId.toString().trim()) {
        studentRow = i + 1;
        break;
      }
    }
    
    if (studentRow === -1) return createResponse(false, 'ไม่พบนักศึกษารหัส: ' + studentId);
    
    var folder = getDocumentsFolder();
    var extension = filename.indexOf('.') !== -1 ? filename.split('.').pop() : (contentType.indexOf('pdf') !== -1 ? 'pdf' : 'jpg');
    var safeFileName = documentType + '_' + (guardianType || '') + '_' + studentId + '_' + new Date().getTime() + '.' + extension;
    
    var decoded = Utilities.base64Decode(fileData);
    var blob = Utilities.newBlob(decoded, contentType, safeFileName);
    var file = folder.createFile(blob);
    
    try {
      file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    } catch (e) {}
    
    var fileId = file.getId();
    var fileUrl = 'https://drive.google.com/uc?export=download&id=' + fileId;
    
    var updated = updateDocumentRecord(studentId, documentType, safeFileName, guardianType, studentRow);
    
    if (updated) {
      clearAppCache();
      logActivity('UPLOAD_DOCUMENT', 'อัปโหลด ' + documentType + (guardianType ? ' (' + guardianType + ')' : '') + ' รหัส ' + studentId, 'system', 'SUCCESS');
      return createResponse(true, 'อัปโหลดเอกสารสำเร็จ', {
        fileUrl: fileUrl,
        previewUrl: 'https://drive.google.com/file/d/' + fileId + '/preview',
        directDownloadUrl: fileUrl,
        webViewUrl: file.getUrl(),
        fileName: safeFileName,
        fileId: fileId
      });
    } else {
      file.setTrashed(true);
      return createResponse(false, 'ไม่สามารถอัปเดตข้อมูลเอกสารในระบบ');
    }
  } catch (error) {
    return createResponse(false, 'เกิดข้อผิดพลาดในการอัปโหลด: ' + error.toString());
  } finally {
    try { lock.releaseLock(); } catch(e) {}
  }
}

function updateDocumentRecord(studentId, documentType, fileName, guardianType, rowNumber) {
  try {
    var sheet = getSheet();
    if (!sheet) return false;
    
    var lastCol = sheet.getLastColumn();
    var rowData = sheet.getRange(rowNumber, 1, 1, lastCol).getValues()[0];
    var headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0];
    
    var updatedIndex = headers.indexOf('updatedAt');
    if (updatedIndex !== -1) rowData[updatedIndex] = new Date().toISOString();
    
    var stdDocs = ['studentIdCard', 'studentHouseReg', 'studentTranscript', 'studentNameChange', 'studentGradCert', 'studentBirthCert'];
    if (stdDocs.indexOf(documentType) !== -1) {
      var colIndex = headers.indexOf(documentType);
      if (colIndex === -1) {
        ensureColumnsExist();
        headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
        colIndex = headers.indexOf(documentType);
      }
      if (colIndex !== -1) rowData[colIndex] = fileName;
    } else if (documentType === 'guardianIdCard' || documentType === 'guardianHouseReg') {
      var colName = documentType === 'guardianIdCard' ? 'guardianIdCardData' : 'guardianHouseRegData';
      var colIndex = headers.indexOf(colName);
      
      if (colIndex === -1) {
        ensureColumnsExist();
        headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
        colIndex = headers.indexOf(colName);
      }
      
      var currentCellVal = rowData[colIndex];
      var guardianData = {
        father: { uploaded: false, fileName: '' },
        mother: { uploaded: false, fileName: '' },
        other: { uploaded: false, fileName: '' }
      };
      
      if (currentCellVal) {
        try {
          var parsed = JSON.parse(currentCellVal);
          if (parsed) guardianData = Object.assign(guardianData, parsed);
        } catch (e) {}
      }
      
      var gType = guardianType || 'father';
      guardianData[gType] = { uploaded: true, fileName: fileName };
      rowData[colIndex] = JSON.stringify(guardianData);
    }
    
    // บันทึกทั้งแถวในคำสั่งเดียว (Single atomic write)
    sheet.getRange(rowNumber, 1, 1, rowData.length).setValues([rowData]);
    SpreadsheetApp.flush();
    clearAppCache();
    return true;
  } catch (error) {
    Logger.log('Error in updateDocumentRecord: ' + error.toString());
    return false;
  }
}

function getDocument(studentId, documentType, guardianType, customFileName) {
  try {
    var fileName = customFileName || '';
    if (fileName) {
      var files = getDocumentsFolder().getFilesByName(fileName);
      if (files.hasNext()) {
        var file = files.next();
        var fileId = file.getId();
        var directDownloadUrl = 'https://drive.google.com/uc?export=download&id=' + fileId;
        return createResponse(true, 'Document found', {
          fileUrl: directDownloadUrl,
          previewUrl: 'https://drive.google.com/file/d/' + fileId + '/preview',
          directDownloadUrl: directDownloadUrl,
          webViewUrl: file.getUrl(),
          fileName: fileName,
          fileId: fileId
        });
      }
    }

    var sheet = getSheet();
    if (!sheet) return createResponse(false, 'Sheet not found');
    
    var data = sheet.getDataRange().getValues();
    var headers = data[0];
    var studentIdIndex = headers.indexOf('studentId');
    
    for (var i = 1; i < data.length; i++) {
      if (data[i][studentIdIndex] && data[i][studentIdIndex].toString().trim() === studentId.toString().trim()) {
        var fileName = '';
        var row = data[i];
        
        var stdDocs = ['studentIdCard', 'studentHouseReg', 'studentTranscript', 'studentNameChange', 'studentGradCert', 'studentBirthCert'];
        if (stdDocs.indexOf(documentType) !== -1) {
          var colIdx = headers.indexOf(documentType);
          if (colIdx !== -1) fileName = row[colIdx];
        } else if (documentType === 'guardianIdCard') {
          var gData = parseGuardianData(row, headers.indexOf('guardianIdCardData'), -1);
          var gType = guardianType || 'father';
          if (gData.guardianIdCard[gType] && gData.guardianIdCard[gType].uploaded) fileName = gData.guardianIdCard[gType].fileName;
        } else if (documentType === 'guardianHouseReg') {
          var gData = parseGuardianData(row, -1, headers.indexOf('guardianHouseRegData'));
          var gType = guardianType || 'father';
          if (gData.guardianHouseReg[gType] && gData.guardianHouseReg[gType].uploaded) fileName = gData.guardianHouseReg[gType].fileName;
        }
        
        if (fileName) {
          var files = getDocumentsFolder().getFilesByName(fileName);
          if (files.hasNext()) {
            var file = files.next();
            var fileId = file.getId();
            var directDownloadUrl = 'https://drive.google.com/uc?export=download&id=' + fileId;
            return createResponse(true, 'Document found', {
              fileUrl: directDownloadUrl,
              previewUrl: 'https://drive.google.com/file/d/' + fileId + '/preview',
              directDownloadUrl: directDownloadUrl,
              webViewUrl: file.getUrl(),
              fileName: fileName,
              fileId: fileId
            });
          }
        }
        return createResponse(false, 'ไม่พบไฟล์เอกสารในระบบ');
      }
    }
    return createResponse(false, 'ไม่พบนักศึกษารหัส: ' + studentId);
  } catch (error) {
    return createResponse(false, 'Error retrieving document: ' + error.toString());
  }
}

function deleteDocument(studentId, documentType, guardianType) {
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(10000);
    var sheet = getSheet();
    if (!sheet) return createResponse(false, 'Sheet not found');
    
    var data = sheet.getDataRange().getValues();
    var headers = data[0];
    var studentIdIndex = headers.indexOf('studentId');
    var updatedIndex = headers.indexOf('updatedAt');
    
    for (var i = 1; i < data.length; i++) {
      if (data[i][studentIdIndex] && data[i][studentIdIndex].toString().trim() === studentId.toString().trim()) {
        var fileName = '';
        var row = i + 1;
        
        if (updatedIndex !== -1) sheet.getRange(row, updatedIndex + 1).setValue(new Date().toISOString());
        
        var stdDocs = ['studentIdCard', 'studentHouseReg', 'studentTranscript', 'studentNameChange', 'studentGradCert', 'studentBirthCert'];
        if (stdDocs.indexOf(documentType) !== -1) {
          var colIdx = headers.indexOf(documentType);
          if (colIdx !== -1) {
            fileName = data[i][colIdx];
            sheet.getRange(row, colIdx + 1).setValue('');
          }
        } else if (documentType === 'guardianIdCard' || documentType === 'guardianHouseReg') {
          var colName = documentType === 'guardianIdCard' ? 'guardianIdCardData' : 'guardianHouseRegData';
          var colIndex = headers.indexOf(colName);
          var gData = parseGuardianData(data[i], headers.indexOf('guardianIdCardData'), headers.indexOf('guardianHouseRegData'));
          var gType = guardianType || 'father';
          
          if (documentType === 'guardianIdCard') {
            if (gData.guardianIdCard[gType]) {
              fileName = gData.guardianIdCard[gType].fileName;
              gData.guardianIdCard[gType] = { uploaded: false, fileName: '' };
              sheet.getRange(row, colIndex + 1).setValue(JSON.stringify(gData.guardianIdCard));
            }
          } else {
            if (gData.guardianHouseReg[gType]) {
              fileName = gData.guardianHouseReg[gType].fileName;
              gData.guardianHouseReg[gType] = { uploaded: false, fileName: '' };
              sheet.getRange(row, colIndex + 1).setValue(JSON.stringify(gData.guardianHouseReg));
            }
          }
        }
        
        if (fileName) {
          try {
            var files = getDocumentsFolder().getFilesByName(fileName);
            while (files.hasNext()) files.next().setTrashed(true);
          } catch (e) {}
        }
        
        clearAppCache();
        logActivity('DELETE_DOCUMENT', 'ลบเอกสาร ' + documentType + (guardianType ? ' (' + guardianType + ')' : '') + ' รหัส ' + studentId, 'admin', 'SUCCESS');
        return createResponse(true, 'ลบเอกสารสำเร็จ');
      }
    }
    return createResponse(false, 'ไม่พบเอกสาร');
  } catch (error) {
    return createResponse(false, 'Error deleting document: ' + error.toString());
  } finally {
    try { lock.releaseLock(); } catch(e) {}
  }
}

    if (!studentId) return createResponse(false, 'กรุณาระบุรหัสนักศึกษา');
    
    var sheet = getSheet();
    if (!sheet) return createResponse(false, 'Sheet not found');
    
    var data = sheet.getDataRange().getValues();
    var headers = data[0];
    var studentIdIndex = headers.indexOf('studentId');
    var updatedIndex = headers.indexOf('updatedAt');
    
    var stdDocCols = [
      'studentIdCard', 'studentHouseReg',
      'studentTranscript', 'studentNameChange', 'studentGradCert', 'studentBirthCert'
    ];
    var gIdIndex = headers.indexOf('guardianIdCardData');
    var gHouseIndex = headers.indexOf('guardianHouseRegData');
    var extraDocsIndex = headers.indexOf('studentExtraDocs');
    
    for (var i = 1; i < data.length; i++) {
      if (data[i][studentIdIndex] && data[i][studentIdIndex].toString().trim() === studentId.toString().trim()) {
        deleteStudentDocuments(studentId);
        var row = i + 1;
        var rowData = sheet.getRange(row, 1, 1, sheet.getLastColumn()).getValues()[0];
        
        for (var d = 0; d < stdDocCols.length; d++) {
          var colIdx = headers.indexOf(stdDocCols[d]);
          if (colIdx !== -1) rowData[colIdx] = '';
        }
        
        if (gIdIndex !== -1) rowData[gIdIndex] = '{"father":{"uploaded":false,"fileName":""},"mother":{"uploaded":false,"fileName":""},"other":{"uploaded":false,"fileName":""}}';
        if (gHouseIndex !== -1) rowData[gHouseIndex] = '{"father":{"uploaded":false,"fileName":""},"mother":{"uploaded":false,"fileName":""},"other":{"uploaded":false,"fileName":""}}';
        if (extraDocsIndex !== -1) rowData[extraDocsIndex] = '[]';
        if (updatedIndex !== -1) rowData[updatedIndex] = new Date().toISOString();
        
        sheet.getRange(row, 1, 1, rowData.length).setValues([rowData]);
        SpreadsheetApp.flush();
        clearAppCache();
        logActivity('CLEAR_ALL_DOCS', 'ลบเอกสารทั้งหมดของนักศึกษารหัส ' + studentId, 'admin', 'SUCCESS');
        return createResponse(true, 'ลบเอกสารทั้งหมดของนักศึกษารหัส ' + studentId + ' เรียบร้อยแล้ว');
      }
    }
    return createResponse(false, 'ไม่พบนักศึกษารหัส: ' + studentId);
  } catch (error) {
    return createResponse(false, 'Error: ' + error.toString());
  } finally {
    try { lock.releaseLock(); } catch(e) {}
  }
}

function generateReport() {
  try {
    var sheet = getSheet();
    if (!sheet) return createResponse(false, 'Sheet not found');
    
    var data = sheet.getDataRange().getValues();
    if (data.length < 2) return createResponse(false, 'ไม่มีข้อมูลนักเรียน');
    
    var headers = data[0];
    var levelIndex = headers.indexOf('educationLevel');
    var majorIndex = headers.indexOf('major');
    var shiftIndex = headers.indexOf('shift');
    var classIndex = headers.indexOf('class');
    var idCardIndex = headers.indexOf('studentIdCard');
    var houseRegIndex = headers.indexOf('studentHouseReg');
    
    var report = {
      totalStudents: data.length - 1,
      byLevel: {},
      byMajor: {},
      byShift: {},
      documentStats: { totalWithIdCard: 0, totalWithHouseReg: 0, completeDocuments: 0 },
      educationLevels: {},
      classDistribution: {}
    };
    
    for (var i = 1; i < data.length; i++) {
      var row = data[i];
      var level = row[levelIndex] || 'ไม่ระบุ';
      var major = row[majorIndex] || 'ไม่ระบุ';
      var shift = row[shiftIndex] || 'ไม่ระบุ';
      var studentClass = row[classIndex] || 'ไม่ระบุ';
      
      report.byLevel[level] = (report.byLevel[level] || 0) + 1;
      report.byMajor[major] = (report.byMajor[major] || 0) + 1;
      report.byShift[shift] = (report.byShift[shift] || 0) + 1;
      report.classDistribution[studentClass] = (report.classDistribution[studentClass] || 0) + 1;
      
      if (!report.educationLevels[level]) report.educationLevels[level] = { total: 0, byMajor: {} };
      report.educationLevels[level].total++;
      report.educationLevels[level].byMajor[major] = (report.educationLevels[level].byMajor[major] || 0) + 1;
      
      if (!!row[idCardIndex]) report.documentStats.totalWithIdCard++;
      if (!!row[houseRegIndex]) report.documentStats.totalWithHouseReg++;
      if (!!row[idCardIndex] && !!row[houseRegIndex]) report.documentStats.completeDocuments++;
    }
    
    return createResponse(true, 'รายงานสรุป', report);
  } catch (error) {
    return createResponse(false, 'Error: ' + error.toString());
  }
}

function getStatistics() {
  try {
    var cache = CacheService.getScriptCache();
    var cached = cache.get('system_statistics');
    if (cached) {
      try { return createResponse(true, 'สถิติระบบ', JSON.parse(cached)); } catch (e) {}
    }
    
    var sheet = getSheet();
    if (!sheet) return createResponse(false, 'Sheet not found');
    
    var data = sheet.getDataRange().getValues();
    if (data.length < 2) {
      return createResponse(true, 'สถิติระบบ', { totalStudents: 0, levels: {}, shifts: {}, majors: {}, documentUploadCount: 0, studentsWithCompleteDocs: 0 });
    }
    
    var headers = data[0];
    var levelIndex = headers.indexOf('educationLevel');
    var majorIndex = headers.indexOf('major');
    var shiftIndex = headers.indexOf('shift');
    var idCardIndex = headers.indexOf('studentIdCard');
    var houseRegIndex = headers.indexOf('studentHouseReg');
    var gIdCardIndex = headers.indexOf('guardianIdCardData');
    var gHouseRegIndex = headers.indexOf('guardianHouseRegData');
    
    var stats = { totalStudents: data.length - 1, levels: {}, shifts: {}, majors: {}, documentUploadCount: 0, studentsWithCompleteDocs: 0 };
    
    for (var i = 1; i < data.length; i++) {
      var row = data[i];
      stats.levels[row[levelIndex] || 'ไม่ระบุ'] = (stats.levels[row[levelIndex] || 'ไม่ระบุ'] || 0) + 1;
      stats.shifts[row[shiftIndex] || 'ไม่ระบุ'] = (stats.shifts[row[shiftIndex] || 'ไม่ระบุ'] || 0) + 1;
      stats.majors[row[majorIndex] || 'ไม่ระบุ'] = (stats.majors[row[majorIndex] || 'ไม่ระบุ'] || 0) + 1;
      
      var uploadedCount = 0;
      if (row[idCardIndex]) uploadedCount++;
      if (row[houseRegIndex]) uploadedCount++;
      
      var guardianData = parseGuardianData(row, gIdCardIndex, gHouseRegIndex);
      if (guardianData.guardianIdCard.father.uploaded) uploadedCount++;
      if (guardianData.guardianIdCard.mother.uploaded) uploadedCount++;
      if (guardianData.guardianIdCard.other.uploaded) uploadedCount++;
      if (guardianData.guardianHouseReg.father.uploaded) uploadedCount++;
      if (guardianData.guardianHouseReg.mother.uploaded) uploadedCount++;
      if (guardianData.guardianHouseReg.other.uploaded) uploadedCount++;
      
      stats.documentUploadCount += uploadedCount;
      if (uploadedCount === 8) stats.studentsWithCompleteDocs++;
    }
    
    try { cache.put('system_statistics', JSON.stringify(stats), CONFIG.CACHE_DURATION_SEC); } catch (e) {}
    
    return createResponse(true, 'สถิติระบบ', stats);
  } catch (error) {
    return createResponse(false, 'Error: ' + error.toString());
  }
}

function importStudentsFromExcel(students) {
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(15000);
    if (!students || !students.length) return createResponse(false, 'ไม่พบข้อมูลที่จะนำเข้า');
    
    var sheet = getSheet();
    if (!sheet) return createResponse(false, 'ไม่พบ Google Sheet');
    
    ensureColumnsExist();
    var headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
    var existingData = sheet.getDataRange().getValues();
    var studentIdIndex = headers.indexOf('studentId');
    var existingIds = [];
    
    for (var i = 1; i < existingData.length; i++) {
      if (existingData[i][studentIdIndex]) existingIds.push(existingData[i][studentIdIndex].toString().trim());
    }
    
    var rowsToAppend = [];
    var imported = 0, skipped = 0;
    var gIdIndex = headers.indexOf('guardianIdCardData');
    var gHouseIndex = headers.indexOf('guardianHouseRegData');
    var createdIndex = headers.indexOf('createdAt');
    var updatedIndex = headers.indexOf('updatedAt');
    var nowIso = new Date().toISOString();
    
    for (var j = 0; j < students.length; j++) {
      var student = students[j];
      var sid = (student.id || student.studentId || '').toString().trim();
      if (!sid || existingIds.indexOf(sid) !== -1) { skipped++; continue; }
      
      var newRow = new Array(headers.length).fill('');
      newRow[headers.indexOf('studentId')] = sid;
      newRow[headers.indexOf('studentName')] = (student.name || student.studentName || '').toString().trim();
      newRow[headers.indexOf('educationLevel')] = (student.level || student.educationLevel || '').toString().trim();
      newRow[headers.indexOf('major')] = (student.major || '').toString().trim();
      newRow[headers.indexOf('class')] = (student.class || '').toString().trim();
      newRow[headers.indexOf('shift')] = (student.shift || student.round || '').toString().trim();
      
      if (gIdIndex !== -1) newRow[gIdIndex] = '{"father":{"uploaded":false,"fileName":""},"mother":{"uploaded":false,"fileName":""},"other":{"uploaded":false,"fileName":""}}';
      if (gHouseIndex !== -1) newRow[gHouseIndex] = '{"father":{"uploaded":false,"fileName":""},"mother":{"uploaded":false,"fileName":""},"other":{"uploaded":false,"fileName":""}}';
      if (createdIndex !== -1) newRow[createdIndex] = nowIso;
      if (updatedIndex !== -1) newRow[updatedIndex] = nowIso;
      
      rowsToAppend.push(newRow);
      existingIds.push(sid);
      imported++;
    }
    
    if (rowsToAppend.length > 0) {
      sheet.getRange(sheet.getLastRow() + 1, 1, rowsToAppend.length, headers.length).setValues(rowsToAppend);
      SpreadsheetApp.flush();
    }
    
    clearAppCache();
    return createResponse(true, 'นำเข้าข้อมูลสำเร็จ ' + imported + ' รายการ', { imported: imported, skipped: skipped });
  } catch (error) {
    return createResponse(false, 'เกิดข้อผิดพลาดในการนำเข้าข้อมูล: ' + error.toString());
  } finally {
    try { lock.releaseLock(); } catch(e) {}
  }
}

function getLogsSheet() {
  try {
    var ss = getSpreadsheet();
    if (!ss) return null;
    var sheet = ss.getSheetByName('Logs');
    if (!sheet) {
      sheet = ss.insertSheet('Logs');
      var headers = ['timestamp', 'action', 'details', 'user', 'status'];
      sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
      sheet.getRange(1, 1, 1, headers.length)
        .setBackground('#1e293b')
        .setFontColor('#ffffff')
        .setFontWeight('bold')
        .setFontFamily('Noto Sans Thai');
      sheet.setFrozenRows(1);
    }
    return sheet;
  } catch(e) {
    return null;
  }
}

function logActivity(action, details, user, status) {
  try {
    var sheet = getLogsSheet();
    if (!sheet) return;
    
    var ts = new Date().toISOString();
    var row = [ts, action || 'GENERAL', details || '', user || 'system', status || 'SUCCESS'];
    sheet.appendRow(row);
    
    // จำกัดจำนวนแถวไม่ให้เกิน 1,000 แถว เพื่อป้องกันชีตบวม
    var lastRow = sheet.getLastRow();
    if (lastRow > 1000) {
      sheet.deleteRows(2, lastRow - 1000);
    }
  } catch(e) {
    Logger.log('Log error: ' + e.toString());
  }
}

function getActivityLogs(params) {
  try {
    var sheet = getLogsSheet();
    if (!sheet) return createResponse(true, 'Logs empty', []);
    
    var data = sheet.getDataRange().getValues();
    if (data.length < 2) return createResponse(true, 'Logs empty', []);
    
    var limit = params && params.limit ? parseInt(params.limit) : 100;
    var filterAction = params && params.actionFilter ? params.actionFilter.trim().toLowerCase() : '';
    
    var logs = [];
    // อ่านย้อนกลับจากแถวล่าสุด (Newest first)
    for (var i = data.length - 1; i >= 1; i--) {
      var row = data[i];
      var action = row[1] || '';
      if (filterAction && action.toLowerCase().indexOf(filterAction) === -1) continue;
      
      logs.push({
        timestamp: row[0],
        action: row[1],
        details: row[2],
        user: row[3],
        status: row[4]
      });
      
      if (logs.length >= limit) break;
    }
    
    return createResponse(true, 'Activity logs retrieved', logs);
  } catch (error) {
    return createResponse(false, 'Error getting activity logs: ' + error.toString());
  }
}

  }
}

// ==================== SYSTEM HEALTH CHECK & AUTO-REPAIR ====================

function performSystemHealthCheck() {
  try {
    var report = {
      timestamp: new Date().toISOString(),
      healthScore: 100,
      status: 'HEALTHY',
      issues: [],
      spreadsheet: {
        connected: false,
        id: getEffectiveSpreadsheetId(),
        name: getEffectiveSheetName(),
        totalStudents: 0,
        columnsValid: false
      },
      drive: {
        connected: false,
        id: getEffectiveFolderId(),
        name: '',
        isPublic: false,
        totalFilesInFolder: 0
      },
      documents: {
        totalRecords: 0,
        existingFilesInDrive: 0,
        brokenLinks: [],
        orphanFiles: []
      }
    };
    
    // 1. ตรวจสอบ Google Sheets
    var sheet = getSheet();
    if (sheet) {
      report.spreadsheet.connected = true;
      var lastCol = sheet.getLastColumn();
      if (lastCol > 0) {
        var headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0];
        var requiredHeaders = ['studentId', 'studentName', 'educationLevel', 'major', 'class', 'shift'];
        var missingHeaders = requiredHeaders.filter(function(h) { return headers.indexOf(h) === -1; });
        report.spreadsheet.columnsValid = missingHeaders.length === 0;
        if (!report.spreadsheet.columnsValid) {
          report.issues.push({ type: 'WARNING', title: 'คอลัมน์ใน Sheet ไม่ครบถ้วน', detail: 'ขาดคอลัมน์: ' + missingHeaders.join(', ') });
          report.healthScore -= 15;
        }
      }
      
      var data = sheet.getDataRange().getValues();
      report.spreadsheet.totalStudents = Math.max(0, data.length - 1);
    } else {
      report.issues.push({ type: 'ERROR', title: 'ไม่สามารถเชื่อมต่อ Google Sheets ได้', detail: 'โปรดตรวจสอบ Spreadsheet ID' });
      report.healthScore -= 40;
    }
    
    // 2. ตรวจสอบ Google Drive Folder
    var folder = null;
    try {
      folder = getDocumentsFolder();
      if (folder) {
        report.drive.connected = true;
        report.drive.name = folder.getName();
        
        try {
          var access = folder.getSharingAccess();
          report.drive.isPublic = (access === DriveApp.Access.ANYONE_WITH_LINK || access === DriveApp.Access.ANYONE);
        } catch(e) {
          report.drive.isPublic = true;
        }
        
        if (!report.drive.isPublic) {
          report.issues.push({ type: 'WARNING', title: 'สิทธิ์โฟลเดอร์ Google Drive ยังไม่เปิดสาธารณะ', detail: 'อาจทำให้ผู้เยี่ยมชมไม่สามารถเปิดดูรูป/PDF ได้' });
          report.healthScore -= 15;
        }
      }
    } catch(errFolder) {
      report.issues.push({ type: 'ERROR', title: 'ไม่สามารถเชื่อมต่อโฟลเดอร์ Google Drive ได้', detail: errFolder.toString() });
      report.healthScore -= 40;
    }
    
    // 3. ตรวจสอบไฟล์ในชีตเทียบกับไฟล์ใน Drive
    if (sheet && folder && data && data.length > 1) {
      var headers = data[0];
      var sidIdx = headers.indexOf('studentId');
      var idCardIdx = headers.indexOf('studentIdCard');
      var houseIdx = headers.indexOf('studentHouseReg');
      var gIdIdx = headers.indexOf('guardianIdCardData');
      var gHouseIdx = headers.indexOf('guardianHouseRegData');
      
      var allExpectedFileNames = {};
      
      for (var i = 1; i < data.length; i++) {
        var row = data[i];
        var sid = row[sidIdx] ? row[sidIdx].toString().trim() : '';
        if (!sid) continue;
        
        if (row[idCardIdx]) allExpectedFileNames[row[idCardIdx].toString().trim()] = { studentId: sid, docType: 'studentIdCard' };
        if (row[houseIdx]) allExpectedFileNames[row[houseIdx].toString().trim()] = { studentId: sid, docType: 'studentHouseReg' };
        
        var gData = parseGuardianData(row, gIdIdx, gHouseIdx);
        var gId = gData.guardianIdCard;
        var gHouse = gData.guardianHouseReg;
        
        if (gId.father && gId.father.uploaded && gId.father.fileName) allExpectedFileNames[gId.father.fileName] = { studentId: sid, docType: 'guardianIdCard (พ่อ)' };
        if (gId.mother && gId.mother.uploaded && gId.mother.fileName) allExpectedFileNames[gId.mother.fileName] = { studentId: sid, docType: 'guardianIdCard (แม่)' };
        if (gId.other && gId.other.uploaded && gId.other.fileName) allExpectedFileNames[gId.other.fileName] = { studentId: sid, docType: 'guardianIdCard (ผู้ปกครองอื่น)' };
        if (gHouse.father && gHouse.father.uploaded && gHouse.father.fileName) allExpectedFileNames[gHouse.father.fileName] = { studentId: sid, docType: 'guardianHouseReg (พ่อ)' };
        if (gHouse.mother && gHouse.mother.uploaded && gHouse.mother.fileName) allExpectedFileNames[gHouse.mother.fileName] = { studentId: sid, docType: 'guardianHouseReg (แม่)' };
        if (gHouse.other && gHouse.other.uploaded && gHouse.other.fileName) allExpectedFileNames[gHouse.other.fileName] = { studentId: sid, docType: 'guardianHouseReg (ผู้ปกครองอื่น)' };
      }
      
      report.documents.totalRecords = Object.keys(allExpectedFileNames).length;
      
      // ดึงรายชื่อไฟล์ทั้งหมดใน Drive Folder
      var driveFilesMap = {};
      var driveFiles = folder.getFiles();
      while (driveFiles.hasNext()) {
        var f = driveFiles.next();
        var fName = f.getName();
        driveFilesMap[fName] = f.getId();
        report.drive.totalFilesInFolder++;
      }
      
      // ตรวจหา Broken Links (มีชื่อในชีตแต่ไม่มีไฟล์ในไดรฟ์)
      for (var expectedName in allExpectedFileNames) {
        if (driveFilesMap[expectedName]) {
          report.documents.existingFilesInDrive++;
        } else {
          report.documents.brokenLinks.push({
            fileName: expectedName,
            studentId: allExpectedFileNames[expectedName].studentId,
            docType: allExpectedFileNames[expectedName].docType
          });
        }
      }
      
      if (report.documents.brokenLinks.length > 0) {
        report.issues.push({
          type: 'WARNING',
          title: 'พบลิงก์เอกสารสูญหาย (Broken Links) ' + report.documents.brokenLinks.length + ' ไฟล์',
          detail: 'มีชื่อเอกสารในชีตแต่ไฟล์ถูกลบออกจาก Google Drive ไปแล้ว'
        });
        report.healthScore -= Math.min(25, report.documents.brokenLinks.length * 3);
      }
      
      // ตรวจหา Orphan Files (มีไฟล์ในไดรฟ์แต่ไม่ได้ผูกกับนักศึกษาคนใด)
      for (var actualName in driveFilesMap) {
        if (!allExpectedFileNames[actualName]) {
          report.documents.orphanFiles.push({
            fileName: actualName,
            fileId: driveFilesMap[actualName]
          });
        }
      }
      
      if (report.documents.orphanFiles.length > 0) {
        report.issues.push({
          type: 'INFO',
          title: 'พบไฟล์ตกค้างใน Google Drive ' + report.documents.orphanFiles.length + ' ไฟล์',
          detail: 'ไฟล์ไม่ได้ผูกกับนักศึกษาในระบบ สามารถกดล้างไฟล์ขยะได้'
        });
      }
    }
    
    report.healthScore = Math.max(0, Math.min(100, report.healthScore));
    if (report.healthScore >= 90) report.status = 'EXCELLENT';
    else if (report.healthScore >= 70) report.status = 'GOOD';
    else if (report.healthScore >= 50) report.status = 'WARNING';
    else report.status = 'CRITICAL';
    
    return createResponse(true, 'System health check completed', report);
  } catch (error) {
    return createResponse(false, 'Health check error: ' + error.toString());
  }
}

function repairSystemIssues(data) {
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(15000);
    var repairType = (data.repairType || data.type || 'all').toLowerCase();
    var results = {
      permissionsFixed: 0,
      brokenLinksCleaned: 0,
      orphansTrashed: 0,
      structureUpdated: false
    };
    
    var folder = getDocumentsFolder();
    var sheet = getSheet();
    
    // 1. ซ่อมแซมสิทธิ์แชร์ไฟล์และโฟลเดอร์
    if (repairType === 'all' || repairType === 'permissions' || repairType === 'sharing') {
      if (folder) {
        try { folder.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); } catch(e) {}
        var files = folder.getFiles();
        while (files.hasNext()) {
          try {
            files.next().setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
            results.permissionsFixed++;
          } catch(e) {}
        }
      }
    }
    
    // 2. ล้างไฟล์ขยะตกค้างใน Drive
    if (repairType === 'all' || repairType === 'orphans' || repairType === 'cleanorphans') {
      if (folder && sheet) {
        var sData = sheet.getDataRange().getValues();
        var allValidFileNames = {};
        if (sData.length > 1) {
          var h = sData[0];
          var sidCol = h.indexOf('studentId');
          var idCCol = h.indexOf('studentIdCard');
          var houseCol = h.indexOf('studentHouseReg');
          var gIdCol = h.indexOf('guardianIdCardData');
          var gHouseCol = h.indexOf('guardianHouseRegData');
          
          for (var r = 1; r < sData.length; r++) {
            var row = sData[r];
            if (!row[sidCol]) continue;
            if (row[idCCol]) allValidFileNames[row[idCCol].toString().trim()] = true;
            if (row[houseCol]) allValidFileNames[row[houseCol].toString().trim()] = true;
            var g = parseGuardianData(row, gIdCol, gHouseCol);
            if (g.guardianIdCard.father?.fileName) allValidFileNames[g.guardianIdCard.father.fileName] = true;
            if (g.guardianIdCard.mother?.fileName) allValidFileNames[g.guardianIdCard.mother.fileName] = true;
            if (g.guardianIdCard.other?.fileName) allValidFileNames[g.guardianIdCard.other.fileName] = true;
            if (g.guardianHouseReg.father?.fileName) allValidFileNames[g.guardianHouseReg.father.fileName] = true;
            if (g.guardianHouseReg.mother?.fileName) allValidFileNames[g.guardianHouseReg.mother.fileName] = true;
            if (g.guardianHouseReg.other?.fileName) allValidFileNames[g.guardianHouseReg.other.fileName] = true;
          }
        }
        
        var dFiles = folder.getFiles();
        while (dFiles.hasNext()) {
          var file = dFiles.next();
          if (!allValidFileNames[file.getName()]) {
            file.setTrashed(true);
            results.orphansTrashed++;
          }
        }
      }
    }
    
    // 3. ปรับปรุงโครงสร้างตาราง Sheet
    if (repairType === 'all' || repairType === 'structure') {
      autoSetupSheetStructure();
      results.structureUpdated = true;
    }
    
    clearAppCache();
    logActivity('SYSTEM_REPAIR', 'ซ่อมแซมระบบสำเร็จ (สิทธิ์: ' + results.permissionsFixed + ', ขยะ: ' + results.orphansTrashed + ')', 'admin', 'SUCCESS');
    return createResponse(true, 'ซ่อมแซมระบบเรียบร้อยแล้ว', results);
  } catch (error) {
    return createResponse(false, 'Repair error: ' + error.toString());
  } finally {
    try { lock.releaseLock(); } catch(e) {}
  }
}

function fixAllFileSharing() {
  return repairSystemIssues({ repairType: 'permissions' });
}

function checkFolderSharing() {
  try {
    var files = getDocumentsFolder().getFiles();
    var fileCount = 0;
    while (files.hasNext()) { files.next(); fileCount++; }
    return createResponse(true, 'Folder sharing status', { folderName: getDocumentsFolder().getName(), totalFiles: fileCount });
  } catch (error) {
    return createResponse(false, 'Error checking folder: ' + error.toString());
  }
}

// ==================== 1-CLICK DATABASE BACKUP & RESTORE SNAPSHOTS ====================

function createBackupSnapshot(data) {
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(15000);
    var ss = getSpreadsheet();
    if (!ss) return createResponse(false, 'Spreadsheet not found');
    
    var sourceSheet = getSheet();
    if (!sourceSheet) return createResponse(false, 'Students sheet not found');
    
    var now = new Date();
    var pad = function(n) { return n < 10 ? '0' + n : '' + n; };
    var timeStr = now.getFullYear() + '' + pad(now.getMonth() + 1) + '' + pad(now.getDate()) + '_' + pad(now.getHours()) + '' + pad(now.getMinutes());
    var backupSheetName = 'Backup_' + timeStr;
    
    // Check if duplicate name exists
    var existingBackup = ss.getSheetByName(backupSheetName);
    if (existingBackup) {
      backupSheetName = 'Backup_' + timeStr + '_' + pad(now.getSeconds());
    }
    
    var backupSheet = sourceSheet.copyTo(ss);
    backupSheet.setName(backupSheetName);
    try { backupSheet.setTabColor('#6366f1'); } catch(e) {}
    
    var studentCount = Math.max(0, backupSheet.getLastRow() - 1);
    
    logActivity('CREATE_BACKUP', 'สร้างจุดสำรองข้อมูล Snapshot: ' + backupSheetName + ' (' + studentCount + ' คน)', 'admin', 'SUCCESS');
    return createResponse(true, 'สร้างจุดสำรองข้อมูลสำเร็จ (' + backupSheetName + ')', {
      sheetName: backupSheetName,
      totalStudents: studentCount,
      timestamp: now.toISOString()
    });
  } catch (error) {
    return createResponse(false, 'Error creating backup: ' + error.toString());
  } finally {
    try { lock.releaseLock(); } catch(e) {}
  }
}

function listBackupSnapshots() {
  try {
    var ss = getSpreadsheet();
    if (!ss) return createResponse(true, 'Backups listed', []);
    
    var sheets = ss.getSheets();
    var backups = [];
    
    for (var i = 0; i < sheets.length; i++) {
      var name = sheets[i].getName();
      if (name.indexOf('Backup_') === 0) {
        var count = Math.max(0, sheets[i].getLastRow() - 1);
        backups.push({
          sheetName: name,
          totalStudents: count,
          createdStr: name.replace('Backup_', '')
        });
      }
    }
    
    // Sort newest first
    backups.sort(function(a, b) {
      return b.sheetName.localeCompare(a.sheetName);
    });
    
    return createResponse(true, 'Backups listed', backups);
  } catch (error) {
    return createResponse(false, 'Error listing backups: ' + error.toString());
  }
}

function restoreBackupSnapshot(data) {
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(20000);
    var ss = getSpreadsheet();
    if (!ss) return createResponse(false, 'Spreadsheet not found');
    
    var targetSheet = getSheet();
    if (!targetSheet) return createResponse(false, 'Target sheet not found');
    
    var backupSheetName = data ? (data.sheetName || data.backupSheetName || '') : '';
    var jsonData = data ? (data.students || data.jsonData || null) : null;
    
    if (backupSheetName) {
      var backupSheet = ss.getSheetByName(backupSheetName);
      if (!backupSheet) return createResponse(false, 'ไม่พบแท็บสำรองข้อมูล: ' + backupSheetName);
      
      var bData = backupSheet.getDataRange().getValues();
      if (bData.length < 1) return createResponse(false, 'แท็บสำรองข้อมูลว่างเปล่า');
      
      targetSheet.clearContents();
      targetSheet.getRange(1, 1, bData.length, bData[0].length).setValues(bData);
      targetSheet.setFrozenRows(1);
      
      var restoredCount = Math.max(0, bData.length - 1);
      clearAppCache();
      logActivity('RESTORE_BACKUP', 'กู้คืนข้อมูลจาก Snapshot: ' + backupSheetName + ' (' + restoredCount + ' คน)', 'admin', 'SUCCESS');
      return createResponse(true, 'กู้คืนข้อมูลจากจุดสำรอง ' + backupSheetName + ' สำเร็จ (' + restoredCount + ' คน)');
    } else if (jsonData && Array.isArray(jsonData) && jsonData.length > 0) {
      // Restore from JSON payload
      return importStudentsFromExcel(jsonData);
    } else {
      return createResponse(false, 'กรุณาระบุแท็บจุดสำรองข้อมูลหรือไฟล์ JSON');
    }
  } catch (error) {
    return createResponse(false, 'Restore error: ' + error.toString());
  } finally {
    try { lock.releaseLock(); } catch(e) {}
  }
}

function deleteBackupSnapshot(data) {
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(10000);
    var ss = getSpreadsheet();
    if (!ss) return createResponse(false, 'Spreadsheet not found');
    
    var backupSheetName = data ? (data.sheetName || data.backupSheetName || '') : '';
    if (!backupSheetName || backupSheetName.indexOf('Backup_') !== 0) {
      return createResponse(false, 'ชื่อแท็บสำรองข้อมูลไม่ถูกต้อง');
    }
    
    var sheetToDelete = ss.getSheetByName(backupSheetName);
    if (!sheetToDelete) return createResponse(false, 'ไม่พบแท็บสำรอง: ' + backupSheetName);
    
    ss.deleteSheet(sheetToDelete);
    logActivity('DELETE_BACKUP', 'ลบแท็บจุดสำรองข้อมูล: ' + backupSheetName, 'admin', 'SUCCESS');
    return createResponse(true, 'ลบแท็บสำรองข้อมูล ' + backupSheetName + ' เรียบร้อยแล้ว');
  } catch (error) {
    return createResponse(false, 'Delete backup error: ' + error.toString());
  } finally {
    try { lock.releaseLock(); } catch(e) {}
  }
}
